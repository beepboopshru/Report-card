Arduino Code Handbook

Grades 6-9 - Y3 Explore Curriculum

52 Arduino UNO sketches

Revised: Interface/Circuit components applied; all LED strips use 30 LEDs.

Using this handbook

Component rule: Each sketch includes only components named in its Interface/Circuit row. Pin Info is used only for pin assignment. BO motors and pumps use the 2-wheel-drive motor driver.
  • Install libraries named in each sketch.
  • LCD address is 0x27; adjust after an I2C scan if required.
  • Calibrate sensor thresholds and use a common ground.

Grade 6

Session 1: RGB LED Strip

Components: RGB LED strip (30 LEDs), push switch
Pin assignment: LED strip D2; push switch D5

// Session 1 - Grade 6: RGB strip
// Arduino UNO

#include <Adafruit_NeoPixel.h>
const byte STRIP_PIN=2,BUTTON_PIN=5,LED_COUNT=30;Adafruit_NeoPixel strip(LED_COUNT,STRIP_PIN,NEO_GRB+NEO_KHZ800);
byte mode=0;bool last=HIGH;unsigned long db=0;
void show(){const uint32_t c[]={strip.Color(0,0,0),strip.Color(255,0,0),strip.Color(0,255,0),strip.Color(0,0,255),strip.Color(255,120,0),strip.Color(150,0,255),strip.Color(0,180,180),strip.Color(255,255,255)};strip.fill(c[mode]);strip.show();}
void setup(){pinMode(BUTTON_PIN,INPUT_PULLUP);strip.begin();strip.clear();show();}
void loop(){bool b=digitalRead(BUTTON_PIN);if(last&& !b&&millis()-db>180){mode=(mode+1)%2;db=millis();show();}last=b;}

Session 4: Ultrasonic Distance System

Components: Ultrasonic sensor, LCD
Pin assignment: ultrasonic D8/D9; LCD A4/A5

// Session 4 - Grade 6: Distance monitor
// Arduino UNO

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27,16,2);const byte TRIG=8,ECHO=9;
float cm(){digitalWrite(TRIG,0);delayMicroseconds(2);digitalWrite(TRIG,1);delayMicroseconds(10);digitalWrite(TRIG,0);unsigned long u=pulseIn(ECHO,HIGH,30000);return u?u*.0343/2:999;}
void setup(){pinMode(TRIG,OUTPUT);pinMode(ECHO,INPUT);lcd.init();lcd.backlight();}
void loop(){float d=cm();bool near=d<20;lcd.setCursor(0,0);lcd.print("Distance: ");lcd.print((int)d);lcd.print(" cm ");lcd.setCursor(0,1);lcd.print(near?"OBJECT NEAR     ":"AREA CLEAR      ");delay(200);}
